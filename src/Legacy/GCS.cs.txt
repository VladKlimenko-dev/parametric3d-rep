public class Param
{
    public enum ParamTarget
    {
        X = 0,
        Y = 1,
        R = 2,
        R1 = 3,
        R2 = 4,
        Other = 5
    }
    public string name;
    private float v;
    public bool reduceable = true;
    public bool changed = false;
    public GLItem2D item;
    public ParamTarget target;
    private Expr exp;

    public float Value
    {
        get { return v; }
        set
        {
            if (v == value) return;
            changed = true;
            v = value;
            switch (target)
            {
                case ParamTarget.X:
                    item.Pos2D = new Vector2(v, item.Pos2D.Y);
                    break;
                case ParamTarget.Y:
                    item.Pos2D = new Vector2(item.Pos2D.X, v);
                    break;
                case ParamTarget.R:
                    if (item is Circle2D)
                    {
                        (item as Circle2D).Radius.Value = v;
                        return;
                    }
                    if (item is Arc2D)
                    {
                        Arc2D A = item as Arc2D;
                        A.Radius.Value = v;
                        for (int i = 1; i < A.child.Count; i++) // Excludint zero-index for Center point
                        {
                            if (A.child[i].name.StartsWith("ArcP"))
                            {
                                Vector2 dir_n = (A.child[i].Pos2D - A.Centre.Pos2D).Normalized();
                                A.child[i].Pos2D = A.Centre.Pos2D + dir_n * v;
                            }
                        }
                  
                        return;
                    }
                    break;
                case ParamTarget.R1:
                    (item as Ellipse2D).Radius1.Value = v;
                    break;
                case ParamTarget.R2:
                    Ellipse2D E = item as Ellipse2D;
                    E.Radius2.Value = v;
                    break;
            }
        }
    }

    public Param(string name, bool reduceable = true)
    {
        this.name = name;
        this.reduceable = reduceable;
        exp = new Expr(this);
    }

    public Param(string name, float value)
    {
        this.name = name;
        this.Value = value;
        exp = new Expr(this);
    }

    public Expr expr
    {
        get
        {
            if (exp == null)
                exp = new Expr(this);
            exp.value = v;
            return exp;
        }

    }

    public Param(GLItem2D _item, ParamTarget _target, string _name, float _value)
    {
        item = _item;
        target = _target;
        this.name = _name;
        this.v = _value;
    }
}

public class Expr
{
    public enum Op
    {
        Undefined,
        Const,
        Param,
        Add,
        Sub,
        Mul,
        Div,
        Sin,
        Cos,
        ACos,
        ASin,
        Sqrt,
        Sqr,
        Atan2,
        Abs,
        Sign,
        Neg,
        Pos,
        Drag,
        Exp
    }

    public Expr a;
    public Expr b;
    public Param param;
    public float value;
    public Op op;

    Expr() { }

    public Expr(float v)
    {
        value = v;
        op = Op.Const;
    }

    internal Expr(Param p)
    {
        param = p;
        op = Op.Param;
    }

    public Expr(Op _op, Expr _a, Expr _b)
    {
        a = _a;
        b = _b;
        op = _op;
    }

    public static readonly Expr zero = new Expr(0.0f);
    public static readonly Expr one = new Expr(1.0f);
    public static readonly Expr mOne = new Expr(-1.0f);
    public static readonly Expr two = new Expr(2.0f);

    public static implicit operator Expr(float v)
    {
        if (v == 0.0) return zero;
        if (v == 1.0) return one;
        Expr result = new Expr();
        result.value = v;
        result.op = Op.Const;
        return result;
    }

    public float Eval()
    {
        switch (op)
        {
            case Op.Const: return value;
            case Op.Param: return param.Value;
            case Op.Add: return a.Eval() + b.Eval();
            case Op.Drag:
            case Op.Sub: return a.Eval() - b.Eval();
            case Op.Mul: return a.Eval() * b.Eval();
            case Op.Div:
                {
                    var bv = b.Eval();
                    if (MathF.Abs(bv) < 1e-10f)
                    {
                        //Debug.Log("Division by zero");
                        bv = 1.0f;
                    }
                    return a.Eval() / bv;
                }
            case Op.Sin: return MathF.Sin(a.Eval());
            case Op.Cos: return MathF.Cos(a.Eval());
            case Op.ACos: return MathF.Acos(a.Eval());
            case Op.ASin: return MathF.Asin(a.Eval());
            case Op.Sqrt: return MathF.Sqrt(a.Eval());
            case Op.Sqr: { float av = a.Eval(); return av * av; }
            case Op.Atan2: return MathF.Atan2(a.Eval(), b.Eval());
            case Op.Abs: return MathF.Abs(a.Eval());
            case Op.Sign: return MathF.Sign(a.Eval());
            case Op.Neg: return -a.Eval();
            case Op.Pos: return a.Eval();
            case Op.Exp: return MathF.Exp(a.Eval());
        }
        return 0.0f;
    }

    public bool IsUnary()
    {
        switch (op)
        {
            case Op.Const:
            case Op.Param:
            case Op.Sin:
            case Op.Cos:
            case Op.ACos:
            case Op.ASin:
            case Op.Sqrt:
            case Op.Sqr:
            case Op.Abs:
            case Op.Sign:
            case Op.Neg:
            case Op.Pos:
            case Op.Exp:
                return true;
        }
        return false;
    }

    public bool IsAdditive()
    {
        switch (op)
        {
            case Op.Drag:
            case Op.Sub:
            case Op.Add:
                return true;
        }
        return false;
    }

    public bool IsZeroConst() { return op == Op.Const && value == 0.0; }
    public bool IsOneConst() { return op == Op.Const && value == 1.0; }
    public bool IsMinusOneConst() { return op == Op.Const && value == -1.0; }
    public bool IsConst() { return op == Op.Const; }
    public bool IsDrag() { return op == Op.Drag; }

    string Quoted()
    {
        if (IsUnary()) return ToString();
        return "(" + ToString() + ")";
    }

    string QuotedAdd()
    {
        if (!IsAdditive()) return ToString();
        return "(" + ToString() + ")";
    }

    public override string ToString()
    {
        switch (op)
        {
            case Op.Const: return value.ToString();
            case Op.Param: return param.name;
            case Op.Add: return a.ToString() + " + " + b.ToString();
            case Op.Sub: return a.ToString() + " - " + b.QuotedAdd();
            case Op.Mul: return a.QuotedAdd() + " * " + b.QuotedAdd();
            case Op.Div: return a.QuotedAdd() + " / " + b.Quoted();
            case Op.Sin: return "sin(" + a.ToString() + ")";
            case Op.Cos: return "cos(" + a.ToString() + ")";
            case Op.ASin: return "asin(" + a.ToString() + ")";
            case Op.ACos: return "acos(" + a.ToString() + ")";
            case Op.Sqrt: return "sqrt(" + a.ToString() + ")";
            case Op.Sqr: return a.Quoted() + " ^ 2";
            case Op.Abs: return "abs(" + a.ToString() + ")";
            case Op.Sign: return "sign(" + a.ToString() + ")";
            case Op.Atan2: return "atan2(" + a.ToString() + ", " + b.ToString() + ")";
            case Op.Neg: return "-" + a.Quoted();
            case Op.Pos: return "+" + a.Quoted();
            case Op.Drag: return a.ToString() + " ≈ " + b.QuotedAdd();
            case Op.Exp: return "exp(" + a.ToString() + ")";
        }
        return "";
    }

    static public Expr operator +(Expr a, Expr b)
    {
        if (a.IsZeroConst()) return b;
        if (b.IsZeroConst()) return a;
        if (b.op == Op.Neg) return a - b.a;
        if (b.op == Op.Pos) return a + b.a;
        return new Expr(Op.Add, a, b);
    }

    static public Expr operator -(Expr a, Expr b)
    {
        if (a.IsZeroConst()) return -b;
        if (b.IsZeroConst()) return a;
        return new Expr(Op.Sub, a, b);
    }

    static public Expr operator *(Expr a, Expr b)
    {
        if (a.IsZeroConst()) return zero;
        if (b.IsZeroConst()) return zero;
        if (a.IsOneConst()) return b;
        if (b.IsOneConst()) return a;
        if (a.IsMinusOneConst()) return -b;
        if (b.IsMinusOneConst()) return -a;
        if (a.IsConst() && b.IsConst()) return a.value * b.value;
        return new Expr(Op.Mul, a, b);
    }

    static public Expr operator /(Expr a, Expr b)
    {
        if (b.IsOneConst()) return a;
        if (a.IsZeroConst()) return zero;
        if (b.IsMinusOneConst()) return -a;
        return new Expr(Op.Div, a, b);
    }

    static public Expr operator -(Expr a)
    {
        if (a.IsZeroConst()) return a;
        if (a.IsConst()) return -a.value;
        if (a.op == Op.Neg) return a.a;
        return new Expr(Op.Neg, a, null);
    }

    static public Expr Sin(Expr x) { return new Expr(Op.Sin, x, null); }
    static public Expr Cos(Expr x) { return new Expr(Op.Cos, x, null); }
    static public Expr ACos(Expr x) { return new Expr(Op.ACos, x, null); }
    static public Expr ASin(Expr x) { return new Expr(Op.ASin, x, null); }
    static public Expr Sqrt(Expr x) { return new Expr(Op.Sqrt, x, null); }
    static public Expr Sqr(Expr x) { return new Expr(Op.Sqr, x, null); }
    static public Expr Abs(Expr x) { return new Expr(Op.Abs, x, null); }
    static public Expr Sign(Expr x) { return new Expr(Op.Sign, x, null); }
    static public Expr Atan2(Expr x, Expr y) { return new Expr(Op.Atan2, x, y); }
    static public Expr Exp(Expr x) { return new Expr(Op.Exp, x, null); }

    Expr d(Param p)
    {
        switch (op)
        {
            case Op.Const: return zero;
            case Op.Param: return (param == p) ? one : zero;
            case Op.Add: return a.d(p) + b.d(p);
            case Op.Drag:
            case Op.Sub: return a.d(p) - b.d(p);
            case Op.Mul: return a.d(p) * b + a * b.d(p);
            case Op.Div: return (a.d(p) * b - a * b.d(p)) / Sqr(b);
            case Op.Sin: return a.d(p) * Cos(a);
            case Op.Cos: return a.d(p) * -Sin(a);
            case Op.ASin: return a.d(p) / Sqrt(one - Sqr(a));
            case Op.ACos: return a.d(p) * mOne / Sqrt(one - Sqr(a));
            case Op.Sqrt: return a.d(p) / (two * Sqrt(a));
            case Op.Sqr: return a.d(p) * two * a;
            case Op.Abs: return a.d(p) * Sign(a);
            case Op.Sign: return zero;
            case Op.Neg: return -a.d(p);
            case Op.Atan2: return (b * a.d(p) - a * b.d(p)) / (Sqr(a) + Sqr(b));
            case Op.Exp: return a.d(p) * Exp(a);
        }
        return zero;
    }

    public void Substitute(Param pa, Param pb)
    {
        if (a != null)
        {
            a.Substitute(pa, pb);
            if (b != null)
            {
                b.Substitute(pa, pb);
            }
        } else
        if (op == Op.Param && param == pa)
        {
            param = pb;
        }
    }

    public void Substitute(Param p, Expr e)
    {
        if (a != null)
        {
            a.Substitute(p, e);
            if (b != null)
            {
                b.Substitute(p, e);
            }
        } else
        if (op == Op.Param && param == p)
        {
            op = e.op;
            a = e.a;
            b = e.b;
            param = e.param;
            value = e.value;
        }
    }

    public bool IsSubstitionForm()
    {
        return op == Op.Sub && a.op == Op.Param && b.op == Op.Param;
    }

    public Param GetSubstitutionParamA()
    {
        if (!IsSubstitionForm()) return null;
        return a.param;
    }

    public Param GetSubstitutionParamB()
    {
        if (!IsSubstitionForm()) return null;
        return b.param;
    }

    public void ReduceParams(List<Param> pars)
    {
        if (op == Op.Param)
        {
            if (param.reduceable && !pars.Contains(param))
            {
                value = Eval();
                op = Op.Const;
                param = null;
            }
            return;
        }

        if (a != null)
        {
            a.ReduceParams(pars);
            if (b != null) b.ReduceParams(pars);
            if (a.IsConst() && (b == null || b.IsConst()))
            {
                value = Eval();
                op = Op.Const;
                a = null;
                b = null;
                param = null;
            }
        }

    }

    public Expr DeepClone()
    {
        Expr result = new Expr();
        result.op = op;
        result.param = param;
        result.value = value;
        if (a != null)
        {
            result.a = a.DeepClone();
            if (b != null)
            {
                result.b = b.DeepClone();
            }
        }
        return result;
    }

    public bool IsDependOn(Param p)
    {
        if (op == Op.Param) return param == p;
        if (a != null)
        {
            if (b != null)
            {
                return a.IsDependOn(p) || b.IsDependOn(p);
            }
            return a.IsDependOn(p);
        }
        return false;
    }

    public Expr Deriv(Param p)
    {
        return d(p);
    }
}

public class ExpVector
{
    public Expr x;
    public Expr y;

    public ExpVector(Expr x, Expr y)
    {
        this.x = x;
        this.y = y;
    }

    public static implicit operator ExpVector(Vector2 v)
    {
        return new ExpVector(v.X, v.Y);
    }

    public static ExpVector operator +(ExpVector a, ExpVector b) { return new ExpVector(a.x + b.x, a.y + b.y); }
    public static ExpVector operator -(ExpVector a, ExpVector b) { return new ExpVector(a.x - b.x, a.y - b.y); }
    public static ExpVector operator *(ExpVector a, ExpVector b) { return new ExpVector(a.x * b.x, a.y * b.y); }
    public static ExpVector operator /(ExpVector a, ExpVector b) { return new ExpVector(a.x / b.x, a.y / b.y); }

    public static ExpVector operator -(ExpVector b) { return new ExpVector(-b.x, -b.y); }

    public static ExpVector operator *(Expr a, ExpVector b) { return new ExpVector(a * b.x, a * b.y); }
    public static ExpVector operator *(ExpVector a, Expr b) { return new ExpVector(a.x * b, a.y * b); }

    public static ExpVector operator /(Expr a, ExpVector b) { return new ExpVector(a / b.x, a / b.y); }
    public static ExpVector operator /(ExpVector a, Expr b) { return new ExpVector(a.x / b, a.y / b); }

    public static Expr Dot(ExpVector a, ExpVector b) { return a.x * b.x + a.y * b.y; }
    public static ExpVector Cross(ExpVector a, ExpVector b)
    {
        return new ExpVector(a.x * b.y - b.x * a.y, 0.0f);
    }

    public static Expr PointLineDistance(ExpVector point, ExpVector l0, ExpVector l1)
    {
        var d = l0 - l1;
        return Cross(d, l0 - point).Magnitude() / d.Magnitude();
    }

    public static float PointLineDistance(Vector2 point, Vector2 l0, Vector2 l1)
    {
        var d = l0 - l1;
        return Vector3.Cross(new Vector3(d.X, d.Y, 0.0f), new Vector3(l0 - point, 0.0f)).Length / d.Length;
    }

    public static ExpVector ProjectPointToLine(ExpVector p, ExpVector l0, ExpVector l1)
    {
        var d = l1 - l0;
        var t = Dot(d, p - l0) / Dot(d, d);
        return l0 + d * t;
    }

    public static Vector2 ProjectPointToLine(Vector2 p, Vector2 l0, Vector2 l1)
    {
        var d = l1 - l0;
        var t = Vector2.Dot(d, p - l0) / Vector2.Dot(d, d);
        return l0 + d * t;
    }

    public Expr Magnitude()
    {
        return Expr.Sqrt(Expr.Sqr(x) + Expr.Sqr(y));
    }

    public ExpVector Normalized()
    {
        return this / Magnitude();
    }

    public Vector2 Eval()
    {
        return new Vector2((float)x.Eval(), (float)y.Eval());
    }

    public bool ValuesEquals(ExpVector o, float eps)
    {
        return MathF.Abs(x.Eval() - o.x.Eval()) < eps &&
               MathF.Abs(y.Eval() - o.y.Eval()) < eps;
    }
}

public static class GaussianMethod
{
    public const double epsilon = 1e-4;
    public const double rankEpsilon = 1e-2;

    public static string Print<T>(this T[,] A)
    {
        string result = "";
        for (int r = 0; r < A.GetLength(0); r++)
        {
            for (int c = 0; c < A.GetLength(1); c++)
            {
                result += A[r, c].ToString() + " ";
            }
            result += "\n";
        }
        return result;
    }

    public static string Print<T>(this T[] A)
    {
        string result = "";
        for (int r = 0; r < A.GetLength(0); r++)
        {
            result += A[r].ToString() + "\n";
        }
        return result;
    }

    public static int Rank(float[,] A)
    {
        var rows = A.GetLength(0);
        var cols = A.GetLength(1);

        int rank = 0;
        float[] rowsLength = new float[rows];

        for (int i = 0; i < rows; i++)
        {
            for (int ii = 0; ii < i; ii++)
            {
                if (rowsLength[ii] <= rankEpsilon) continue;

                float sum = 0;
                for (int j = 0; j < cols; j++)
                {
                    sum += A[ii, j] * A[i, j];
                }
                for (int j = 0; j < cols; j++)
                {
                    A[i, j] -= A[ii, j] * sum / rowsLength[ii];
                }
            }

            float len = 0;
            for (int j = 0; j < cols; j++)
            {
                len += A[i, j] * A[i, j];
            }
            if (len > rankEpsilon)
            {
                rank++;
            }
            rowsLength[i] = len;
        }

        return rank;
    }

    public static void Solve(float[,] A, float[] B, ref float[] X)
    {
        var rows = A.GetLength(0);
        var cols = A.GetLength(1);
        float t = 0.0f;

        for (int r = 0; r < rows; r++)
        {

            var mr = r;
            float max = 0.0f;
            for (int rr = r; rr < rows; rr++)
            {
                if (MathF.Abs(A[rr, r]) <= max) continue;
                max = MathF.Abs(A[rr, r]);
                mr = rr;
            }

            if (max < epsilon) continue;

            for (int c = 0; c < cols; c++)
            {
                t = A[r, c];
                A[r, c] = A[mr, c];
                A[mr, c] = t;
            }

            t = B[r];
            B[r] = B[mr];
            B[mr] = t;

            // normalize
            /*
			double scale = A[r, r];
			for(int c = 0; c < cols; c++) {
				A[r, c] /= scale;
			}
			B[r] /= scale;
			*/

            // 
            for (int rr = r + 1; rr < rows; rr++)
            {
                float coef = A[rr, r] / A[r, r];
                for (int c = 0; c < cols; c++)
                {
                    A[rr, c] -= A[r, c] * coef;
                }
                B[rr] -= B[r] * coef;
            }
        }

        for (int r = rows - 1; r >= 0; r--)
        {
            if (Math.Abs(A[r, r]) < epsilon) continue;
            float xx = B[r] / A[r, r];
            for (int rr = rows - 1; rr > r; rr--)
            {
                xx -= X[rr] * A[r, rr] / A[r, r];
            }
            X[r] = xx;
        }
    }

}

public class EquationSystem
{
    public List<Expr> equations = new List<Expr>();

    public enum SolveResult
    {
        OKAY,
        DIDNT_CONVEGE,
        REDUNDANT,
        POSTPONE
    }

    bool isDirty = true;

    public bool IsDirty { get { return isDirty; } }
    public int maxSteps = 20;
    public int dragSteps = 3;
    public bool revertWhenNotConverged = true;

    Expr[,] J;
    float[,] A;
    float[,] AAT;
    float[] B;
    float[] X;
    float[] Z;
    float[] oldParamValues;

    List<Expr> sourceEquations = new List<Expr>();
    List<Param> parameters = new List<Param>();

    List<Param> currentParams = new List<Param>();

    Dictionary<Param, Param> subs;

    public EquationSystem() { }

    public void AddEquation(Expr eq)
    {
        sourceEquations.Add(eq);
        isDirty = true;
    }

    public void AddEquation(ExpVector v)
    {
        sourceEquations.Add(v.x);
        sourceEquations.Add(v.y);
        isDirty = true;
    }

    public void AddEquations(List<Expr> eq)
    {
        sourceEquations.AddRange(eq);
        isDirty = true;
    }

    public void RemoveEquation(Expr eq)
    {
        sourceEquations.Remove(eq);
        isDirty = true;
    }

    public void AddParameter(Param p)
    {
        //if (!parameters.Contains(p))
        {
            parameters.Add(p);
            isDirty = true;
        }
    }

    public void AddParameters(List<Param> p)
    {
        for (int i = 0; i < p.Count; i++)
            //if (!parameters.Contains(p[i]))
            {
                parameters.Add(p[i]);
                isDirty = true;
            }
    }

    public void RemoveParameter(Param p)
    {
        parameters.Remove(p);
        isDirty = true;
    }

    Dictionary<Param, Param> SolveBySubstitution()
    {
        var subs = new Dictionary<Param, Param>();

        for (int i = 0; i < equations.Count; i++)
        {
            var eq = equations[i];
            if (!eq.IsSubstitionForm()) continue;
            var a = eq.GetSubstitutionParamA();
            var b = eq.GetSubstitutionParamB();
            if (Math.Abs(a.Value - b.Value) > GaussianMethod.epsilon) continue;
            if (!currentParams.Contains(b))
            {
                var t = a;
                a = b;
                b = t;
            }
            // TODO: Check errors
            //if(!parameters.Contains(b)) {
            //	continue;
            //}

            foreach (var k in subs.Keys.ToList())
            {
                if (subs[k] == b)
                {
                    subs[k] = a;
                }
            }
            subs[b] = a;
            equations.RemoveAt(i--);
            currentParams.Remove(b);

            for (int j = 0; j < equations.Count; j++)
            {
                equations[j].Substitute(b, a);
            }
        }
        return subs;
    }

    void BackSubstitution(Dictionary<Param, Param> subs)
    {
        if (subs == null) return;
        for (int i = 0; i < parameters.Count; i++)
        {
            var p = parameters[i];
            if (!subs.ContainsKey(p)) continue;
            p.Value = subs[p].Value;
        }
    }

    void UpdateDirty()
    {
        if (isDirty)
        {
            equations = sourceEquations.Select(e => e.DeepClone()).ToList();
            currentParams = parameters.ToList();
            /*
			foreach(var e in equations) {
				e.ReduceParams(currentParams);
			}*/
            //currentParams = parameters.Where(p => equations.Any(e => e.IsDependOn(p))).ToList();
            subs = SolveBySubstitution();

            J = WriteJacobian(equations, currentParams);
            A = new float[J.GetLength(0), J.GetLength(1)];
            B = new float[equations.Count];
            X = new float[currentParams.Count];
            Z = new float[A.GetLength(0)];
            AAT = new float[A.GetLength(0), A.GetLength(0)];
            oldParamValues = new float[parameters.Count];
            isDirty = false;
            dofChanged = true;
        }
    }

    public bool HasDragged()
    {
        return equations.Any(e => e.IsDrag());
    }

    public void EvalJacobian(Expr[,] J, ref float[,] A, bool clearDrag)
    {
        UpdateDirty();
        for (int r = 0; r < J.GetLength(0); r++)
        {
            if (clearDrag && equations[r].IsDrag())
            {
                for (int c = 0; c < J.GetLength(1); c++)
                {
                    A[r, c] = 0.0f;
                }
                continue;
            }
            for (int c = 0; c < J.GetLength(1); c++)
            {
                A[r, c] = J[r, c].Eval();
            }
        }
    }

    void StoreParams()
    {
        for (int i = 0; i < parameters.Count; i++)
        {
            oldParamValues[i] = parameters[i].Value;
        }
    }

    void RevertParams()
    {
        for (int i = 0; i < parameters.Count; i++)
        {
            parameters[i].Value = oldParamValues[i];
        }
    }

    public void Eval(ref float[] B, bool clearDrag)
    {
        for (int i = 0; i < equations.Count; i++)
        {
            if (clearDrag && equations[i].IsDrag())
            {
                B[i] = 0.0f;
                continue;
            }
            B[i] = equations[i].Eval();
        }
    }

    public bool IsConverged(bool checkDrag)
    {
        for (int i = 0; i < equations.Count; i++)
        {
            if (!checkDrag && equations[i].IsDrag())
                continue;
            if (Math.Abs(B[i]) < GaussianMethod.epsilon)
                continue;
            return false;
        }
        return true;
    }

    static Expr[,] WriteJacobian(List<Expr> equations, List<Param> parameters)
    {
        var J = new Expr[equations.Count, parameters.Count];
        for (int r = 0; r < equations.Count; r++)
        {
            var eq = equations[r];
            for (int c = 0; c < parameters.Count; c++)
            {
                var u = parameters[c];
                J[r, c] = eq.Deriv(u);
            }
        }
        return J;
    }

    public void SolveLeastSquares(float[,] A, float[] B, ref float[] X)
    {

        // A^T * A * X = A^T * B
        var rows = A.GetLength(0);
        var cols = A.GetLength(1);

        var time = Environment.TickCount;
        for (int r = 0; r < rows; r++)
        {
            for (int c = 0; c < rows; c++)
            {
                float sum = 0.0f;
                for (int i = 0; i < cols; i++)
                {
                    if (A[c, i] == 0 || A[r, i] == 0) continue;
                    sum += A[r, i] * A[c, i];
                }
                AAT[r, c] = sum;
            }
        }
        //Debug.Log("AAT time " + (Time.realtimeSinceStartup - time) * 1000);

        GaussianMethod.Solve(AAT, B, ref Z);

        for (int c = 0; c < cols; c++)
        {
            float sum = 0.0f;
            for (int r = 0; r < rows; r++)
            {
                sum += Z[r] * A[r, c];
            }
            X[c] = sum;
        }

    }

    public void Clear()
    {
        parameters.Clear();
        currentParams.Clear();
        equations.Clear();
        sourceEquations.Clear();
        isDirty = true;
        UpdateDirty();
    }

    public bool TestRank(out int dof)
    {
        EvalJacobian(J, ref A, clearDrag: false);
        int rank = GaussianMethod.Rank(A);
        dof = A.GetLength(1) - rank;
        return rank == A.GetLength(0);
    }

    public string stats { get; private set; }
    public bool dofChanged { get; private set; }

    public SolveResult Solve()
    {
        dofChanged = false;
        UpdateDirty();
        StoreParams();
        int steps = 0;
        do
        {
            bool isDragStep = steps <= dragSteps;
            Eval(ref B, clearDrag: !isDragStep);
            if (IsConverged(checkDrag: isDragStep))
            {
                if (steps > 0)
                    dofChanged = true;
                BackSubstitution(subs);
                return SolveResult.OKAY;
            }
            EvalJacobian(J, ref A, clearDrag: !isDragStep);
            SolveLeastSquares(A, B, ref X);
            for (int i = 0; i < currentParams.Count; i++)
            {
                currentParams[i].Value -= X[i];
            }
        } while (steps++ <= maxSteps);
        IsConverged(checkDrag: false);
        if (revertWhenNotConverged)
        {
            RevertParams();
            dofChanged = false;
        }
        return SolveResult.DIDNT_CONVEGE;
    }
}

public abstract class Constraint
{
    public List<GLItem2D> items = new List<GLItem2D>();
    public eConstraint type;
    public int id;
    public Sketch? sketch;
    public bool isDestroyed = false;

    protected List<Param> ps = new List<Param>();
    protected List<Expr> es = new List<Expr>();

    public List<Param> parameters
    {
        get
        {
            return ps;
        }
    }
    public List<Expr> equations
    {
        get
        {
            return es;
        }
    }

    public Constraint(Sketch sk)
    {
        id = Project.GetNewID();
        sketch = sk;
        sk.AddConstraint(this);
    }

    public virtual void BuildEquationsAndParameters() { }
}

public class CHorizontal : Constraint
{
    public CHorizontal(Sketch sk) : base(sk) { type = eConstraint.Horizontal; }

    public CHorizontal(Sketch sk, Point2D p0, Point2D p1) : base(sk)
    {
        items.Add(p0);
        items.Add(p1);
        type = eConstraint.Horizontal;
    }

    public CHorizontal(Sketch sk, Line2D line) : base(sk)
    {
        items.Add(line.P0);
        items.Add(line.P1);
        type = eConstraint.Horizontal;
    }

    public override void BuildEquationsAndParameters()
    {
        Point2D p0 = items[0] as Point2D;
        Point2D p1 = items[1] as Point2D;
        ps = [p0.Y, p1.Y];
        es = [p0.Y.expr - p1.Y.expr];
    }
}

public class CCoincident : Constraint
{
    public CCoincident(Sketch sk) : base(sk) { type = eConstraint.Coincident; }

    public CCoincident(Sketch sk, GLItem2D l0, GLItem2D l1) : base(sk)
    {
        items.Add(l0);
        items.Add(l1);
        type = eConstraint.Coincident;
    }

    public override void BuildEquationsAndParameters()
    {
        var i0 = items[0];
        var i1 = items[1];
        // Point to line - coincidence:
        if ((i0.type == eItemType.Point2D && i1.type == eItemType.Line2D) || (i0.type == eItemType.Point2D && i1.type == eItemType.Line2D))
        {
            Point2D p = i0.type == eItemType.Point2D ? i0 as Point2D : i1 as Point2D;
            Line2D L = i1.type == eItemType.Line2D ? i1 as Line2D : i0 as Line2D;
            // (lp1x - lp0x) * (py - lp0y) - (lp1y - lp0y) * (px - lp0x)
            Expr e = (L.P1.X.expr - L.P0.X.expr) * (p.Y.expr - L.P0.Y.expr) - (L.P1.Y.expr - L.P0.Y.expr) * (p.X.expr - L.P0.X.expr);

            ps = [p.X, p.Y];
            es = [e];
            return;
        }

        if ((i0.type == eItemType.Point2D && i1.type == eItemType.Curved) || (i1.type == eItemType.Point2D && i0.type == eItemType.Curved))
        {
            Point2D p = i0.type == eItemType.Point2D ? i0 as Point2D : i1 as Point2D;
            Curved2D C2d = i1.type == eItemType.Curved ? i1 as Curved2D : i0 as Curved2D;
            switch (i1.subtype)
            {
                case eSubType.Circle2D:
                    {
                        Circle2D C = C2d as Circle2D;
                        Vector2 v = Geometry.ProjectPointOnCircle2D(p.Pos2D, C.Centre.Pos2D, C.Radius.Value);
                        ps = [p.X, p.Y];
                        es = [p.X.expr - v.X, p.Y.expr - v.Y];
                    }
                    break;
                case eSubType.Arc2D_cp:
                case eSubType.Arc2D_ppp:
                    {
                        Arc2D A = C2d as Arc2D;
                        Vector2 v = Geometry.ProjectPointOnCircle2D(p.Pos2D, A.Centre.Pos2D, A.Radius.Value);
                        ps = [p.X, p.Y];
                        es = [p.X.expr - v.X, p.Y.expr - v.Y];
                    }
                    break;
            }
        }

        if (i0.type == eItemType.Point2D && i1.type == eItemType.Point2D)
        {
            Point2D p1 = i0 as Point2D;
            Point2D p2 = i1 as Point2D;
            ps = [p1.X, p1.Y, p2.X, p2.Y];
            Expr eq1 = p1.X.expr - p2.X.expr;
            Expr eq2 = p1.Y.expr - p2.Y.expr;
            es = [eq1, eq2];
        }
    }
}

public class CTangent : Constraint
{
    public CTangent(Sketch sk) : base(sk) { type = eConstraint.Tangent; }

    public CTangent(Sketch sk, GLItem2D l0, GLItem2D l1) : base(sk)
    {
        items.Add(l0);
        items.Add(l1);
        type = eConstraint.Tangent;
    }

    public override void BuildEquationsAndParameters()
    {
        var i0 = items[0];
        var i1 = items[1];

        if ((i0.type == eItemType.Line2D && i1.type == eItemType.Curved) || (i1.type == eItemType.Line2D && i0.type == eItemType.Curved))
        {
            Line2D L = i0.type == eItemType.Line2D? i0 as Line2D : i1 as Line2D;
            Curved2D C2d = i1.type == eItemType.Curved ? i1 as Curved2D : i0 as Curved2D;
            if (i1.subtype == eSubType.Circle2D)
            {
                Circle2D C = C2d as Circle2D;
                // Get coordinates of the point on line:
                Expr dir_x = L.P0.X.expr - L.P1.X.expr;
                Expr dir_y = L.P0.Y.expr - L.P1.Y.expr;
                Expr t = ((C.Centre.X.expr - L.P0.X.expr) * dir_x + (C.Centre.Y.expr - L.P0.Y.expr) * dir_y) / (dir_x * dir_x + dir_y * dir_y);
                Expr Lp_x = L.P0.X.expr + t * dir_x;
                Expr Lp_y = L.P0.Y.expr + t * dir_y;
                // Get coordinates of the point on cicrcle
                Expr dir_pc_x = Lp_x - C.Centre.X.expr;
                Expr dir_pc_y = Lp_y - C.Centre.Y.expr;
                Expr llen2 = dir_pc_x * dir_pc_x + dir_pc_y * dir_pc_y;
                Expr llen = new Expr(Expr.Op.Sqrt, llen2, null);
                Expr e1 = llen - C.Radius.expr;
                ps = [C.Radius];
                es = [e1];
                return;
            }
        }
    }
}

public class CDistance : Constraint
{
    public CDistance(Sketch sk) : base(sk) { type = eConstraint.Distance; }

    public CDistance(Sketch sk, GLItem2D i0, GLItem2D i1) : base(sk)
    {
        items.Add(i0);
        items.Add(i1);
        type = eConstraint.Distance;
    }

    public override void BuildEquationsAndParameters()
    {
        var i0 = items[0];
        var i1 = items[1];

        // Point - Point distance:
        if (i0.type == eItemType.Point2D && i1.type == eItemType.Point2D)
        {
            Point2D p1 = i0 as Point2D;
            Point2D p2 = i1 as Point2D;
            ps = [p1.X, p1.Y, p2.X, p2.Y];
            Expr eq1 = p1.X.expr - p2.X.expr;
            Expr eq2 = p1.Y.expr - p2.Y.expr;
            es = [eq1, eq2];
        }

        // Point - Line distance:
        if ((i0.type == eItemType.Point2D && i1.type == eItemType.Line2D) || (i0.type == eItemType.Point2D && i1.type == eItemType.Line2D))
        {
            Point2D p = i0.type == eItemType.Point2D ? i0 as Point2D : i1 as Point2D;
            Line2D L = i1.type == eItemType.Line2D ? i1 as Line2D : i0 as Line2D;
            Vector2 v = Geometry.ProjectPointOnLine2D(L.P0.Pos2D, L.P1.Pos2D, i0.Pos2D, out float t);
            Param pvx = new Param(L, Param.ParamTarget.X, "px", v.X);
            Param pvy = new Param(L, Param.ParamTarget.Y, "py", v.Y);
            ps = [pvx, pvy, p.X, p.Y];
            Expr eq1 = new Expr(Expr.Op.Sub, p.X.expr, pvx.expr);
            Expr eq2 = new Expr(Expr.Op.Sub, p.Y.expr, pvy.expr);
            es = [eq1, eq2];
        }

        // Line - Curve distance:
        if ((i0.type == eItemType.Line2D && i1.type == eItemType.Curved) || (i1.type == eItemType.Line2D && i0.type == eItemType.Curved))
        {
            Line2D L = i0.type == eItemType.Line2D ? i0 as Line2D : i1 as Line2D;
            Curved2D C2d = i1.type == eItemType.Curved ? i1 as Curved2D : i0 as Curved2D;
            if (i1.subtype == eSubType.Circle2D)
            {
                Circle2D C = C2d as Circle2D;
                // Get coordinates of the point on line:
                Expr dir_x = L.P0.X.expr - L.P1.X.expr;
                Expr dir_y = L.P0.Y.expr - L.P1.Y.expr;
                Expr t = ((C.Centre.X.expr - L.P0.X.expr) * dir_x + (C.Centre.Y.expr - L.P0.Y.expr) * dir_y) / (dir_x * dir_x + dir_y * dir_y);
                Expr Lp_x = L.P0.X.expr + t * dir_x;
                Expr Lp_y = L.P0.Y.expr + t * dir_y;
                // Get coordinates of the point on cicrcle
                Expr dir_pc_x = Lp_x - C.Centre.X.expr;
                Expr dir_pc_y = Lp_y - C.Centre.Y.expr;
                Expr llen2 = dir_pc_x * dir_pc_x + dir_pc_y * dir_pc_y;
                Expr llen = new Expr(Expr.Op.Sqrt, llen2, null);
                Expr e1 = llen - C.Radius.expr;
                ps = [C.Radius];
                es = [e1];
                return;
            }
        }
    }
